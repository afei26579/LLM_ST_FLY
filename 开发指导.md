## 项目概述

### 🎯 项目定位
LLM ST FLY 是一个现代化的全栈智能管理系统，集成了多种AI功能和企业级系统管理能力。项目采用前后端分离架构，定位为企业级AI应用平台。

### ✨ 核心特性
- **多模态AI集成**: 文本对话、图像生成、音频处理、视频生成、文档阅读
- **智能体系统**: 专业化的AI助手（旅游、诗词、客服等）
- **企业级权限**: 完整的用户、角色、权限管理体系
- **实时通信**: 支持流式输出的聊天系统
- **多租户支持**: 用户隔离的数据管理
- **主题系统**: 支持多种UI主题切换

#### 3. API服务层架构

前端采用分层的API服务设计，`services/` 目录包含所有与后端交互的服务：

```typescript
// services/api.ts - 主API服务类
class APIService {
  private instance: AxiosInstance
  
  constructor() {
    this.instance = axios.create({
      baseURL: 'http://localhost:8000/api/v1/',
      timeout: 30000,
    })
    
    // 请求拦截器 - 添加认证token
    this.instance.interceptors.request.use((config) => {
      const token = localStorage.getItem('token')
      if (token) {
        config.headers.Authorization = `Bearer ${token}`
      }
      return config
    })
    
    // 响应拦截器 - 统一错误处理和数据格式化
    this.instance.interceptors.response.use(
      (response) => {
        // 统一处理成功响应
        return response.data
      },
      (error) => {
        // 统一错误处理
        if (error.response?.status === 401) {
          // 处理认证失败，跳转到登录页
          router.push('/login')
        }
        return Promise.reject(error)
      }
    )
  }
  
  // 用户认证相关API
  async login(credentials: LoginForm): Promise<LoginResponse> {
    return this.instance.post('/auth/login/', credentials)
  }
  
  async getUserProfile(): Promise<UserProfile> {
    return this.instance.get('/auth/users/me/')
  }
  
  // 聊天相关API
  async sendMessage(data: SendMessageRequest): Promise<ChatResponse> {
    return this.instance.post('/chat/send/', data)
  }
  
  async getConversations(params?: ConversationParams): Promise<ConversationList> {
    return this.instance.get('/chat/conversations/', { params })
  }
  
  // AI功能相关API
  async generateImage(data: ImageGenerationRequest): Promise<ImageResponse> {
    return this.instance.post('/ai-image/generate/', data)
  }
  
  async analyzeDocument(file: File): Promise<DocumentAnalysis> {
    const formData = new FormData()
    formData.append('file', file)
    return this.instance.post('/ai-reading/analyze/', formData, {
      headers: { 'Content-Type': 'multipart/form-data' }
    })
  }
}

// 创建API服务实例供全局使用
```

```typescript
router.beforeEach((to, from, next) => {
  if (to.meta.requiresAuth) {
    const token = localStorage.getItem('token')
    if (!token) {
      next({ name: 'login', query: { redirect: to.fullPath } })
      return
    }
    
    // 角色权限检查
    if (to.meta.requiredRole) {
      const userData = JSON.parse(localStorage.getItem('userInfo') || '{}')
      if (userData?.role !== to.meta.requiredRole && userData?.role !== 'admin') {
        next('/')
        return
      }
    }
  }
  next()
})
```

```python
class ChatService:
    def __init__(self):
        self.dashscope_api_key = settings.DASHSCOPE_API_KEY
    
    def send_message(self, user, conversation_id, message_content, enable_thinking=False):
        """发送消息并获取AI回复"""
        try:
            # 保存用户消息
            user_message = self.save_user_message(conversation_id, message_content)
            
            # 构建对话历史
            messages = self.build_conversation_history(conversation_id)
            
            # 调用AI API
            if enable_thinking:
                response = self.call_thinking_api(messages)
            else:
                response = self.call_standard_api(messages)
            
            # 保存AI回复
            ai_message = self.save_ai_message(conversation_id, response)
            
            return {
                'user_message': user_message,
                'ai_message': ai_message,
                'conversation': conversation_id
            }
            
        except Exception as e:
            logger.error(f"聊天服务错误: {e}")
            raise ChatServiceError(str(e))
    
    def call_thinking_api(self, messages):
        """调用支持思考过程的API"""
        response = dashscope.Generation.call(
            model='qwen-max',
            messages=messages,
            enable_thinking=True,
            stream=True
        )
        return self.process_thinking_response(response)
```

```python
# chat/ai-image/services.py - 图像生成服务
class ImageGenerationService:
    def generate_image(self, user, prompt, **params):
        """生成图像"""
        # 创建生成任务
        task = ImageGenerationTask.objects.create(
            user=user,
            task_id=str(uuid.uuid4()),
            prompt=prompt,
            size=params.get('size', '1024*1024'),
            style=params.get('style'),
            status='pending'
        )
        
        try:
            # 调用DashScope图像生成API
            response = dashscope.ImageSynthesis.call(
                model='wanx-v1',
                prompt=prompt,
                negative_prompt=params.get('negative_prompt'),
                size=task.size,
                n=params.get('n', 1),
                style=params.get('style'),
            )
            
            if response.status_code == HTTPStatus.OK:
                # 处理成功响应
                images = self.process_generated_images(task, response)
                task.status = 'completed'
                task.save()
                
                # 更新用户统计
                self.update_user_stats(user)
                
                return {
                    'task': task,
                    'images': images
                }
            else:
                task.status = 'failed'
                task.error_message = response.message
                task.save()
                raise ImageGenerationError(response.message)
                
        except Exception as e:
            task.status = 'failed'
            task.error_message = str(e)
            task.save()
            raise
```

```python
# chat/ai-reading/services.py - 文档分析服务
class DocumentAnalysisService:
    def analyze_document(self, user, file_content, filename):
        """分析上传的文档"""
        # 上传文件到AI服务
        file_object = self.upload_file_to_ai_service(file_content, filename)
        
        # 创建文档记录
        document = Document.objects.create(
            user=user,
            name=filename,
            size=len(file_content),
            file_type=self.get_file_type(filename),
            file_object_id=file_object.id,
            last_modified=int(time.time())
        )
        
        # 执行文档分析
        analysis_result = self.perform_document_analysis(file_object)
        
        # 保存分析结果
        analysis = DocumentAnalysis.objects.create(
            document=document,
            summary=analysis_result['summary'],
            key_points=analysis_result['key_points'],
            keywords=analysis_result['keywords'],
            entities=analysis_result['entities'],
            suggested_questions=analysis_result['suggested_questions']
        )
        
        return {
            'document': document,
            'analysis': analysis
        }
    
    def qa_with_document(self, document_id, question):
        """基于文档进行问答"""
        document = Document.objects.get(id=document_id)
        
        # 构建问答提示词
        context = self.get_document_context(document)
        prompt = f"基于以下文档内容回答问题:\n\n{context}\n\n问题: {question}"
        
        # 调用AI API
        response = self.call_qa_api(prompt)
        
        # 保存问答历史
        qa_history = QAHistory.objects.create(
            document=document,
            question=question,
            answer=response['answer'],
            confidence_score=response.get('confidence', 0.0)

        return qa_history
```

```python
class Permission(models.Model):
    """权限模型"""

    content_type = models.ForeignKey(ContentType, on_delete=models.CASCADE)
    
class Role(models.Model):
    """角色模型"""
    code = models.CharField(max_length=20, unique=True)
    name = models.CharField(max_length=50)
    permissions = models.ManyToManyField(Permission)
    is_system = models.BooleanField(default=False)
    
class User(AbstractUser):
    """用户模型"""
    role = models.ForeignKey(Role, on_delete=models.SET_NULL, null=True)
    
    def has_permission(self, permission_code):
        """检查用户是否有特定权限"""
        if self.is_superuser:
            return True
        if self.role:
            return self.role.permissions.filter(codename=permission_code).exists()
        return False
```

#### 2. 权限装饰器
```python
# users/permissions.py - 权限装饰器
from functools import wraps
from rest_framework.permissions import BasePermission

class HasPermission(BasePermission):
    """自定义权限类"""
    def __init__(self, permission_code):
        self.permission_code = permission_code
    
    def has_permission(self, request, view):
        return request.user.has_permission(self.permission_code)

def require_permission(permission_code):
    """权限检查装饰器"""
    def decorator(view_func):
        @wraps(view_func)
        def wrapper(request, *args, **kwargs):
            if not request.user.has_permission(permission_code):
                return Response({'error': '权限不足'}, status=403)
            return view_func(request, *args, **kwargs)
        return wrapper
    return decorator
```

```typescript
// router/index.ts - 路由权限
interface RouteMeta {
  requiresAuth: boolean
  requiredRole?: string
  requiredPermissions?: string[]
}

router.beforeEach(async (to, from, next) => {
  const authStore = useAuthStore()
  
  if (to.meta.requiresAuth && !authStore.isAuthenticated) {
    next({ name: 'login', query: { redirect: to.fullPath } })
    return
  }
  
  if (to.meta.requiredPermissions) {
    const hasPermissions = await authStore.checkPermissions(to.meta.requiredPermissions)
    if (!hasPermissions) {
      next('/unauthorized')
      return
    }
  }
  
  next()
})
```
